検査に使ったコマンド:

```sh
# (C1) 変更内容の確認
git diff README.md
sed -n '1,3p;14,22p' README.md

# (C2) CLAUDE.md の各単位が README に残っていないか + 対照で検査の有効性を確認
python3 - <<'EOF'
import io, re
cm = io.open('claude/CLAUDE.md', encoding='utf-8').read()
rm = io.open('README.md', encoding='utf-8').read()
units = []
for line in cm.splitlines():
    line = re.sub(r'^[-*]\s+', '', line).strip()
    if not line:
        continue
    for s in re.split(r'(?<=[.?!])\s+', line):
        s = s.strip()
        if len(s) >= 12:
            units.append(s)
scan = lambda t: [u for u in units if u in t]
print('units:', len(units))
print('hits in README:', scan(rm))                      # -> []
print('hits in control:', scan(rm + chr(10) + units[0]))  # -> 先頭文を検出
EOF

# (C3) 訳語経由の写し
for w in 一時ファイル バックアップを残さ 日本語 専門用語 推測 小さく テストが通 結論 根拠 選択肢 削除 パス; do
  printf '%-14s %s\n' "$w" "$(grep -c -- "$w" README.md)"
done

# (C4) 節と全体の分量
sed -n '14,22p' README.md | wc -lm   # 9 行 / 342 文字(削除前: 9 行 / 362 文字)
wc -lm < README.md                   # 57 行 / 1965 文字
git show 21c4d8a:README.md | sed -n '14,22p' | wc -lm
```
